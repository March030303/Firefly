# 后端准备

前端向后端发送请求，拿取数据。所以先用Express（基于nodejs的web开发框架）

1、下载express
```
npm i express
```

2、在js文件中使用express
```
1、引入express

const express = require('express');

2、创建应用对象

const app = express();

```

3、启动express
```
node ???.js
```

# Ajax

首先，需要一个后端的接口，也就是在express搭建好框架后，在server,js中：
```
//1、引入express

const express = require('express');

  

//2、创建应用对象

const app = express();

  

let data = {

    name:'张三'

}

data = JSON.stringify(data);

  

//3、创建路由规则

app.all('/nodejs',(req,res) => {

res.setHeader('Access-Control-Allow-Origin','*');

    res.send(data);

})

  

app.post('/nodejs',(req,res) => {

res.setHeader('Access-Control-Allow-Origin','*');

    res.send();

  

})

app.get('/delay',(req,res)=>{

    res.setHeader('Access-Control-Allow-Origin','*');

    setTimeout(()=>{

        res.send('延时');

    },5000);

})

app.all('/axios',(req,res)=>{

const data ={

    name:'张三'

}

res.setHeader('Access-Control-Allow-Origin','*');

res.setHeader('Access-Control-Allow-Methods','GET,POST,OPTIONS');  // 加上这个，就可以解决options请求了

res.setHeader('Access-Control-Allow-Headers','Content-Type');        // 加上这个,自定义请求头就可以了

res.send(JSON.stringify(data));


})

//4、监听端口启动服务

app.listen(3000,()=>{

    console.log('服务启动成功');

})
```

后端接口准备好后，就可以get了：
```
//1、创建对象

        const xhr = new XMLHttpRequest();

        //2、初始化：设置请求方法和url

        xhr.open('GET','http://127.0.0.1:3000/nodejs')//这里有接口

        //3、发送请求

        xhr.send();

        //4、事件绑定，处理服务端返回的结果

        xhr.onreadystatechange = function(){

            //readystate是xhr对象的属性,表示状态0 1 2 3 4

            if(xhr.readyState===4){

                //再判断状态码

                if(xhr.status>=200&&xhr.status<300){

                    //处理结果 行 头 体

                    //1、行

                    console.log(xhr.status);//状态码

                    console.log(xhr.statusText);//状态码字符串

  

                    //2、头

                    console.log(xhr.getAllResponseHeaders());

                    //3、体

                    console.log(xhr.responseText);

                    //4、将结果展示在页面上

                    const result = document.querySelector('#result');

                    result.innerHTML = xhr.responseText;

                }else{

                    console.log('请求失败');

            }
```
